set(MODEL_ASSET_PACK "${CMAKE_SOURCE_DIR}/build/metal-baseline/models/FlightHelmet.dmodel" CACHE FILEPATH "Prepared static glTF model")
set(MODEL_BUNDLE "${CMAKE_CURRENT_BINARY_DIR}/model-viewer.app")
file(MAKE_DIRECTORY "${MODEL_BUNDLE}/Contents/Resources")
configure_file(Info.plist "${MODEL_BUNDLE}/Contents/Info.plist" COPYONLY)
configure_file(model.metal "${MODEL_BUNDLE}/Contents/Resources/model.metal" COPYONLY)
configure_file(ASSET-NOTICE.md "${MODEL_BUNDLE}/Contents/Resources/ASSET-NOTICE.md" COPYONLY)
if (EXISTS "${MODEL_ASSET_PACK}")
    configure_file("${MODEL_ASSET_PACK}" "${MODEL_BUNDLE}/Contents/Resources/FlightHelmet.dmodel" COPYONLY)
endif()
add_darling_executable(model-viewer main.mm renderer.mm)
target_compile_options(model-viewer PRIVATE -fobjc-arc -fblocks)
set_target_properties(model-viewer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON
    RUNTIME_OUTPUT_DIRECTORY "${MODEL_BUNDLE}/Contents/MacOS")
target_link_libraries(model-viewer Metal MetalKit AppKit QuartzCore Foundation CoreGraphics cxx)
